export const meta = {
  name: 'plan47-design-wave',
  description: 'Plan 47 design: five lanes, each designed by one model family, reviewed by another, then answered',
  phases: [
    { title: 'Design', detail: 'one design pass per lane in its own worktree' },
    { title: 'Review', detail: 'cross-family review of each lane' },
    { title: 'Answer', detail: 'the designing family fixes or rebuts each finding' },
  ],
}

const BASE = args.base
const B = '/tmp/plan47-briefs'
const R = '/tmp/plan47-reports'
const LANES = [
  { key: 'A', brief: 'A-annotations.md', branch: 'p47/a-annotations', designer: { kind: 'relay', model: 'gpt-6.1-sol' }, reviewers: [{ kind: 'claude', name: 'opus' }, { kind: 'relay', model: 'gpt-6-astra', name: 'astra' }] },
  { key: 'E', brief: 'E-epics.md', branch: 'p47/e-epics', designer: { kind: 'relay', model: 'gpt-6-astra' }, reviewers: [{ kind: 'claude', name: 'opus' }] },
  { key: 'G', brief: 'G-diagrams.md', branch: 'p47/g-diagrams', designer: { kind: 'relay', model: 'gpt-6.1-sol' }, reviewers: [{ kind: 'claude', name: 'opus' }] },
  { key: 'S', brief: 'S-slices.md', branch: 'p47/s-slices', designer: { kind: 'claude' }, reviewers: [{ kind: 'relay', model: 'gpt-6.1-sol', name: 'sol' }] },
  { key: 'R', brief: 'R-roadmap.md', branch: 'p47/r-roadmap', designer: { kind: 'claude' }, reviewers: [{ kind: 'relay', model: 'gpt-6-astra', name: 'astra' }] },
]

const jobReturn = (text) => Object.fromEntries(
  String(text ?? '').split('\n').flatMap((line) => {
    const m = /^([a-z-]+): (.*)$/.exec(line.trim())
    return m ? [[m[1], m[2]]] : []
  }),
)

const wt = (lane) => `/tmp/plan47-wt/${lane.key}`

async function run(who, lane, task, { label, phase, mode }) {
  if (who.kind === 'relay') {
    const header = [`model: ${who.model}`, 'effort: xhigh', `mode: ${mode}`, `cd: ${wt(lane)}`].join('\n')
    const r = jobReturn(await agent(`${header}\n---\n${task}`, { agentType: 'relay:job', label, phase }))
    if (r.state !== 'completed' || r.undelivered) return { ok: false, state: r.state, error: r.error, job: r.job, report: r.report }
    return { ok: true, job: r.job, report: r.report }
  }
  const text = await agent(
    `${task}\n\nYou are a Claude agent: run every shell command with \`cd ${wt(lane)} && ...\`, and use absolute paths under ${wt(lane)} for every file you read or edit. Never edit /Users/darkomijic/.t3/worktrees/software-delivery-protocol/feature-roadmap-and-slices-v0.`,
    { agentType: 'general-purpose', label, phase },
  )
  return text ? { ok: true, text } : { ok: false, state: 'no answer' }
}

const results = await pipeline(
  LANES,
  async (lane) => {
    const task = `Your lane brief is ${B}/${lane.brief}. Read ${B}/common.md first, then the lane brief, and carry out the lane's design pass.\n\nWorktree: ${wt(lane)} (branch ${lane.branch}), base commit ${BASE}. Write your report to ${R}/${lane.key}-design.md and return the same text as your final answer.`
    const design = await run(lane.designer, lane, task, { label: `design-${lane.key}`, phase: 'Design', mode: 'yolo' })
    log(`lane ${lane.key} design: ${design.ok ? 'done' : 'FAILED ' + design.state}`)
    return { lane: lane.key, design }
  },
  async (prev, lane) => {
    if (!prev.design.ok) return prev
    const reviews = await parallel(lane.reviewers.map((who) => async () => {
      const path = `${R}/${lane.key}-review-${who.name}.md`
      const task = `Review lane ${lane.key} of plan 47. Your instructions are ${B}/review.md; read them first.\n\nLane brief: ${B}/${lane.brief}. Designer's report: ${R}/${lane.key}-design.md. Worktree: ${wt(lane)} (branch ${lane.branch}), base commit ${BASE}. Report path: ${path}.`
      const r = await run(who, lane, task, { label: `review-${lane.key}-${who.name}`, phase: 'Review', mode: 'read-only' })
      return { reviewer: who.name, path: who.kind === 'relay' ? r.report : path, ...r }
    }))
    log(`lane ${lane.key} reviews: ${reviews.filter((r) => r && r.ok).length}/${lane.reviewers.length}`)
    return { ...prev, reviews: reviews.filter(Boolean) }
  },
  async (prev, lane) => {
    if (!prev.design.ok || !prev.reviews.some((r) => r.ok)) return prev
    const reviewPaths = prev.reviews.filter((r) => r.ok).map((r) => `- ${r.reviewer}: ${r.path}`).join('\n')
    const task = `Answer the reviews of lane ${lane.key} of plan 47. Your instructions are ${B}/address.md; read them first.\n\nLane brief: ${B}/${lane.brief}. Designer's report: ${R}/${lane.key}-design.md. Reviews:\n${reviewPaths}\n\nWorktree: ${wt(lane)} (branch ${lane.branch}), base commit ${BASE}. Write your report to ${R}/${lane.key}-answer.md and return the same text.`
    const answer = await run(lane.designer, lane, task, { label: `answer-${lane.key}`, phase: 'Answer', mode: 'yolo' })
    log(`lane ${lane.key} answer: ${answer.ok ? 'done' : 'FAILED ' + answer.state}`)
    return { ...prev, answer }
  },
)

return results.map((r) => r && ({
  lane: r.lane,
  design: { ok: r.design.ok, job: r.design.job, state: r.design.state },
  reviews: (r.reviews ?? []).map((v) => ({ reviewer: v.reviewer, ok: v.ok, path: v.path, state: v.state })),
  answer: r.answer ? { ok: r.answer.ok, job: r.answer.job, state: r.answer.state } : null,
}))
